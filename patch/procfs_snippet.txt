    ## monitor
    (
        set +e
        echo "time_ns, mem_bytes, cpu_user_usec, cpu_sys_usec"
        start=$(date +%s%6N)

        # function to get all PIDs: task + children recursively
        get_all_pids() {
            local pid=$1
            echo "$pid"
            for child in $(pgrep -P "$pid" 2>/dev/null || true); do
                get_all_pids "$child"
            done
        }

        while kill -0 $task 2>/dev/null; do
            now=$(date +%s%6N)
            elapsed_us=$((now - start))

            total_mem=0
            total_cpu_user=0
            total_cpu_sys=0

            for pid in $(get_all_pids "$task"); do
                # memory in bytes: VmRSS from /proc/<pid>/status
                mem=$(awk '/VmRSS/{print $2*1024}' /proc/$pid/status 2>/dev/null || echo 0)
                total_mem=$((total_mem + mem))

                # CPU: user and system time in clock ticks from /proc/<pid>/stat
                # fields 14 = utime, 15 = stime
                read utime stime < <(awk '{print $14, $15}' /proc/$pid/stat 2>/dev/null || echo "0 0")
                utime=${utime:-0}
                stime=${stime:-0}

                total_cpu_user=$((total_cpu_user + utime))
                total_cpu_sys=$((total_cpu_sys + stime))
            done

            echo "$elapsed_us, $mem, $total_cpu_user, $total_cpu_sys"
            sleep 0.1
        done
    ) > bblog_deep.log &



safe_num() {
    case $1 in
        ''|*[!0-9]*) echo 0 ;;
        *) echo "$1" ;;
    esac
}
